local function UpdateSelectedStages(val)

    local selectedMap = {}

    local selectedList = {}

    if type(val) == "table" then

        for k, v in pairs(val) do

            if type(k) == "number" and type(v) == "string" then

                selectedMap[v] = true

                table.insert(selectedList, v)

            elseif type(k) == "string" and v == true then

                selectedMap[k] = true

                table.insert(selectedList, k)

            elseif type(v) == "string" then

                selectedMap[v] = true

                table.insert(selectedList, v)

            end

        end

    elseif type(val) == "string" then

        selectedMap[val] = true

        table.insert(selectedList, val)

    end



    -- Automatically map "ด่านท้ายสุด" / "Last Stage" to Stage 13

    for item, enabled in pairs(selectedMap) do

        if enabled and (item:find("ด่านท้าย") or item:find("Last") or item:find("End")) then

            selectedMap["Stage 13"] = true

            selectedMap["Stage13"] = true

        end

    end



    Config.SelectedStagesMap = selectedMap

    Config.SelectedStagesList = selectedList

end

        if isfile and isfile(CONFIG_FILE) and readfile then

            local raw = readfile(CONFIG_FILE)

            local data = HttpService:JSONDecode(raw)

            if type(data) == "table" then

                if data.IsFarmRunning ~= nil then

                    Config.IsFarmRunning = data.IsFarmRunning

                end

                if type(data.SelectedRaritiesList) == "table" and #data.SelectedRaritiesList > 0 then

                    -- Trust the saved list verbatim: no rarity is re-injected on load.
                    local savedRarities = {}

local function IsInDungeon()

    if game.PlaceId == 135412798316370 then

        return true

    end

    if LocalPlayer and LocalPlayer:GetAttribute("DungeonArena") == true then

        return true

    end

    local pg = LocalPlayer and LocalPlayer:FindFirstChild("PlayerGui")

    local dHud = pg and pg:FindFirstChild("DungeonHud")

    local bottom = dHud and dHud:FindFirstChild("Bottom")

    if dHud and dHud:IsA("ScreenGui") and dHud.Enabled

        and bottom and bottom:IsA("GuiObject") and bottom.Visible then

        return true

    end

    return false

-- Removed the FlingVelocity bypass: the game's FlingGuardController reads that
-- child as "already being flung" and switches itself off. Staying at the official
-- walk speed means the guard never fights us, so we leave its protection on.



local function RecoverFromRagdoll(char)

    char = char or LocalPlayer.Character

    if not char then return end



    if RagdollModule and RagdollModule.UnRagdoll then

        pcall(function() RagdollModule.UnRagdoll(char) end)

    end



    local hum = char:FindFirstChildOfClass("Humanoid")

    local hrp = char:FindFirstChild("HumanoidRootPart")

    if hum and hum.Health > 0 then

        hum:SetStateEnabled(Enum.HumanoidStateType.GettingUp, true)

        hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)

        hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)

        pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)

    end



    for _, ch in ipairs(char:GetDescendants()) do

        if ch.Name == "RagdollSocket" or ch.Name == "RagdollAttachment" then

            pcall(function() ch:Destroy() end)

        elseif ch:IsA("Motor6D") and not ch.Enabled then

            ch.Enabled = true

        end

    end



    if hrp then

        hrp.CanCollide = false

        local currentRot = hrp.CFrame - hrp.CFrame.Position

        hrp.CFrame = CFrame.new(hrp.Position.X, math.max(hrp.Position.Y, DEFAULT_GROUND_Y), hrp.Position.Z) * currentRot

        hrp.AssemblyLinearVelocity = Vector3.zero

    end



    char:SetAttribute("Ragdolled", nil)

    char:SetAttribute("RagdollUntil", nil)

end



local function SetupProtection(char)

    if not char then return end



    local hrp = char:WaitForChild("HumanoidRootPart", 5)

    local hum = char:WaitForChild("Humanoid", 5)

    if hrp then hrp.CanCollide = false end



    char:GetAttributeChangedSignal("Ragdolled"):Connect(function()

        if char:GetAttribute("Ragdolled") == true then

            RecoverFromRagdoll(char)

        end

    end)



    if hum then

        hum.StateChanged:Connect(function(oldState, newState)

            if newState == Enum.HumanoidStateType.Physics 

               or newState == Enum.HumanoidStateType.Ragdoll 

               or newState == Enum.HumanoidStateType.FallingDown then

                RecoverFromRagdoll(char)

            end

        end)

    end

end

local _lastSpeedCheck = 0

local _cachedSpeedStat = 0

local _cachedExpected = 35



local function GetSafeMaxSpeed()

    local now = os.clock()

    if now - _lastSpeedCheck > 0.3 then

        _lastSpeedCheck = now

        if not PlayerData then

            pcall(function() PlayerData = require(ReplicatedStorage.Shared.Modules.DataReceiver.PlayerData) end)

        end

        if not SpeedConfig then

            pcall(function() SpeedConfig = require(ReplicatedStorage.Shared.Configs.SpeedConfig) end)

        end

        if PlayerData and PlayerData.Get then

            _cachedSpeedStat = tonumber(PlayerData.Get("Speed")) or 0

        end

        if SpeedConfig and SpeedConfig.ToWalkSpeed then

            local slowMode = PlayerData and PlayerData.Get and PlayerData.Get(SpeedConfig.SlowModeDataName) == true or false

            _cachedExpected = SpeedConfig.ToWalkSpeed(_cachedSpeedStat, slowMode)

        end

    end



    -- Same budget the game's own SpeedController applies: official curve x carry
    -- slowdown. Anything above this gets position-corrected by the server, which is
    -- the rubberband. No multiplier on top.
    local carryMult = 1

    if ServerData and ServerData.Get and AdminAbuseConfig and AdminAbuseConfig.GetCarrySpeedMultiplier then

        local s = ServerData.Get("Stealing")

        local stealingData = type(s) == "table" and s[tostring(LocalPlayer.UserId)]

        if type(stealingData) == "table" and stealingData.Heavy then

            carryMult = AdminAbuseConfig.GetCarrySpeedMultiplier(stealingData.Weight)

        end

    end



    local safeSpeed = _cachedExpected * carryMult

    if carryMult < 1 and AdminAbuseConfig and AdminAbuseConfig.Rocket and AdminAbuseConfig.Rocket.CarrySpeed then

        safeSpeed = math.max(safeSpeed, AdminAbuseConfig.Rocket.CarrySpeed.MinimumWalkSpeed)

    end

    return math.max(math.floor(safeSpeed), 8), _cachedSpeedStat, _cachedExpected

end



local function GetMultiEnemyAvoidance(curPos)

    local packsFolder = Workspace:FindFirstChild("Game") and Workspace.Game:FindFirstChild("Packs")

    if not packsFolder then return Vector3.zero end



    local avoid = Vector3.zero

    for _, stage in ipairs(packsFolder:GetChildren()) do

        local enemy = stage:FindFirstChild("Enemy")

        local primary = enemy and (enemy.PrimaryPart or enemy:FindFirstChildWhichIsA("BasePart", true))

        if primary then

            local ePos = primary.Position

            local deltaX = curPos.X - ePos.X

            local deltaZ = curPos.Z - ePos.Z

            local dist2D = math.sqrt(deltaX * deltaX + deltaZ * deltaZ)



            if dist2D < 25 and dist2D > 0.1 then

                local dodgeDir = (ePos.Z > CORRIDOR_Z_CENTER) and -0.8 or 0.8

                avoid = avoid + Vector3.new(0, 0, dodgeDir)

            end

        end

    end

    return avoid

end



local function GroundGlideTo(targetPos, stopDistance, stageName, speedOverride)

    stopDistance = stopDistance or 2.5

    local maxSafe, curStat, expectedWS = GetSafeMaxSpeed()

    local speed = speedOverride and math.min(speedOverride, maxSafe) or maxSafe



    local hrp, hum = GetCharParts()

    if not hrp or not hum or hum.Health <= 0 then return false end



    StopGroundMovement()



    local initialDist = (Vector3.new(targetPos.X, 0, targetPos.Z) - Vector3.new(hrp.Position.X, 0, hrp.Position.Z)).Magnitude

    local dynamicTimeout = math.max((initialDist / math.max(speed, 25)) + 8, 20)

    local t0 = os.clock()

    local reached = false



    activeGroundConn = RunService.Heartbeat:Connect(function()

        if not Config.IsFarmRunning or not hrp or not hrp.Parent or hum.Health <= 0 then

            reached = true

            StopGroundMovement()

            return

        end



        if IsInDungeon and IsInDungeon() then

            reached = true

            StopGroundMovement()

            return

        end



        if LocalPlayer.Character and LocalPlayer.Character:GetAttribute("Ragdolled") == true then

            RecoverFromRagdoll(LocalPlayer.Character)

        end



        local curPos = hrp.Position



        -- Never fall through floor anywhere on the map

        if curPos.Y < 68.5 then

            hrp.CFrame = CFrame.new(curPos.X, DEFAULT_GROUND_Y, curPos.Z)

            hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, 0, hrp.AssemblyLinearVelocity.Z)

        end



        local delta2D = Vector3.new(targetPos.X - curPos.X, 0, targetPos.Z - curPos.Z)

        local dist2D = delta2D.Magnitude



        if dist2D <= stopDistance or (targetPos == STAGING_POSITION and curPos.X <= 1155.0 and dist2D <= 8.0) or (os.clock() - t0 > dynamicTimeout) then

            reached = true

            StopGroundMovement()

            return

        end



        -- Cruise at full speed, brake with the remaining distance so a fast
        -- character never overshoots the stop radius and oscillates.
        speed = math.min(GetSafeMaxSpeed(), math.max(dist2D * 10, 8))

        local dir2D = delta2D.Unit



        -- Smooth Corridor wall deflection & full enemy avoidance (No CFrame teleport snaps = Zero Rubberbanding!)

        if IsInCorridor(curPos) then

            local evadeVec = GetMultiEnemyAvoidance(curPos)

            if evadeVec.Magnitude > 0 then

                local testZ = curPos.Z + (dir2D.Z + evadeVec.Z) * 6

                if testZ > (CORRIDOR_Z_MAX - 8) or testZ < (CORRIDOR_Z_MIN + 8) then

                    evadeVec = Vector3.new(0, 0, -evadeVec.Z)

                end

                dir2D = (dir2D + evadeVec).Unit

            end



            if curPos.Z > (CORRIDOR_Z_MAX - 6) and dir2D.Z > 0 then

                dir2D = Vector3.new(dir2D.X, 0, -0.6).Unit

            elseif curPos.Z < (CORRIDOR_Z_MIN + 6) and dir2D.Z < 0 then

                dir2D = Vector3.new(dir2D.X, 0, 0.6).Unit

            end

        end



        hum.WalkSpeed = speed

        hum:Move(dir2D, false)

        -- The engine's PlayerModule rewrites MoveDirection every frame, so
        -- Humanoid:Move alone never sticks. The velocity write is what actually
        -- moves us — kept at the game's own walk speed so nothing corrects it.

        local currentYVel = math.max(hrp.AssemblyLinearVelocity.Y, -8)

        hrp.AssemblyLinearVelocity = Vector3.new(dir2D.X * speed, currentYVel, dir2D.Z * speed)

    end)

    getgenv()._AnimeGroundConn = activeGroundConn



    while not reached and Config.IsFarmRunning and hrp and hum.Health > 0 and (os.clock() - t0 <= dynamicTimeout) do

        task.wait(0.02)

    end



    StopGroundMovement()

    task.wait(0.04)

    return true

end

        if Remotes and Remotes.Get then

            if exitRemote then

                exitRemote:Fire()

            end

        end

    end)

        hum.Jump = true

        hum:ChangeState(Enum.HumanoidStateType.Jumping)

    end

    local t0 = os.clock()

        task.wait(0.05)

    local hrp2 = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

            if primary then

                return primary.Position

            end

        end

    end



    return Vector3.new(1140.98, 70.0, -261.86)

local lastUpgradeAttempt = 0

local isUpgradingTreadmill = false



local function TryUpgradeTreadmill(force)

    if not Config.AutoUpgradeTreadmill and not force then return false end

    if isUpgradingTreadmill then return false end



    -- Strict condition: Only proceed if money is strictly reached

    local canUp, nextPrice, curLevelName = CanUpgradeTreadmill()

    if not canUp then

        return false

    end



    if os.clock() - lastUpgradeAttempt < 2.5 then return false end



    local hrp, hum = GetCharParts()

    if not hrp or not hum then return false end



    -- Server requires player to be near their plot's treadmill button

    local isAtBase = hrp.Position.X <= STAGE_CORRIDOR_ENTRY_X

    if not isAtBase and not force then return false end



    isUpgradingTreadmill = true

    lastUpgradeAttempt = os.clock()



    local wasOnTm = IsOnTreadmill()

    if wasOnTm then

        ExitTreadmill()

        task.wait(0.15)

    end



    local btnPos = GetPlayerTreadmillUpgradePosition()

    local priceFormatted = NumberFormatter and NumberFormatter.FormatCompact(nextPrice) or tostring(nextPrice)

-- 10. MULTI-STAGE TARGET FINDER (WITH STREAMING SUPPORT & BLACKLIST)

local TargetBlacklist = {}



local function FindBestTargetInActiveStages()

    local candidates = {}

    local packsRoot = Workspace:FindFirstChild("Game") and Workspace.Game:FindFirstChild("Packs")

    if not packsRoot then return nil end



    local maxStage = 13

    if packsRoot then

        for _, child in ipairs(packsRoot:GetChildren()) do

            local sNum = tonumber(string.match(child.Name, "%d+"))

            if sNum and sNum > maxStage then

                maxStage = sNum

            end

        end

    end



    for stageIdx = 1, maxStage do

        local stageName = "Stage" .. stageIdx

        if IsStageEnabled(stageName) then

            local stageFolder = packsRoot:FindFirstChild(stageName)

            local spawnedFolder = stageFolder and stageFolder:FindFirstChild("SpawnedPacks")

            if spawnedFolder then

                for _, pack in ipairs(spawnedFolder:GetChildren()) do

                    if pack.Name ~= "PackAreaRock" then

                        -- Check blacklist timeout (prevents repeatedly targeting stuck/glitched packs)

                        local blacklistedAt = TargetBlacklist[pack]

                        if not blacklistedAt or (os.clock() - blacklistedAt > 10) then

                            local odd = 1

                            local rarity = "Common"

                            if CharacterConfig and CharacterConfig.Get then

                                local charData = CharacterConfig.Get(pack.Name)

                                if charData then

                                    odd = charData.NumericalOdd or 1

                                    rarity = charData.Rarity or "Unknown"

                                end

                            end



                            local hasAnyRarityFilter = false

                            for _, v in pairs(Config.TargetRarities) do

                                if v == true then hasAnyRarityFilter = true break end

                            end



                            local isAllowed = not hasAnyRarityFilter

                                or (Config.TargetRarities["All"] == true)

                                or (Config.TargetRarities["ทุกระดับ (All Rarities)"] == true)

                                or (Config.TargetRarities[rarity] == true)



                            if isAllowed then

                                local prompt = pack:FindFirstChild("StealPrompt", true) 

                                    or pack:FindFirstChildWhichIsA("ProximityPrompt", true)



                                local packPos = nil

                                if prompt and prompt.Parent then

                                    if prompt.Parent:IsA("Attachment") then

                                        packPos = prompt.Parent.WorldPosition

                                    elseif prompt.Parent:IsA("BasePart") then

                                        packPos = prompt.Parent.Position

                                    end

                                end



                                if not packPos then

                                    local sphere = pack:FindFirstChild("Sphere")

                                    if sphere and sphere:IsA("BasePart") then

                                        packPos = sphere.Position

                                    elseif pack.PrimaryPart then

                                        packPos = pack.PrimaryPart.Position

                                    else

                                        local ok, pPos = pcall(function() return pack:GetPivot().Position end)

                                        if ok and pPos then

                                            packPos = pPos

                                        end

                                    end

                                end



                                if packPos then

                                    table.insert(candidates, {

                                        Pack = pack,

                                        Name = pack.Name,

                                        Stage = stageName,

                                        Odd = odd,

                                        Rarity = rarity,

                                        Position = packPos,

                                        Prompt = prompt

                                    })

                                end

                            end

                        end

                    end

                end

            end

        end

    end



    if #candidates == 0 then return nil end



    local hrp = GetCharParts()

    local myPos = hrp and hrp.Position or STAGING_POSITION



    -- เรียงตามระยะทางจากตัวละคร (ใกล้สุดก่อน)

    table.sort(candidates, function(a, b)

        local distA = (Vector3.new(a.Position.X, 0, a.Position.Z) - Vector3.new(myPos.X, 0, myPos.Z)).Magnitude

        local distB = (Vector3.new(b.Position.X, 0, b.Position.Z) - Vector3.new(myPos.X, 0, myPos.Z)).Magnitude

        return distA < distB

    end)



    return candidates[1]

end

local lastSetTeamAt = 0

-- Shared throttle: the server drops SetTeam calls less than ~0.5s apart with
-- "Slow down!", and two engines (or a manual click) used to hit it together.
local function SetTeam(ids)

    if type(ids) ~= "table" or #ids == 0 then return false, nil end

    if not Remotes then return false, nil end

    local waitLeft = 0.6 - (os.clock() - lastSetTeamAt)

    if waitLeft > 0 then task.wait(waitLeft) end

    local lastRes = nil

    for attempt = 1, 2 do

        lastSetTeamAt = os.clock()

        local ok, res = pcall(function()

            return Remotes.Get("DungeonParty", "SetTeam"):Invoke(ids)

        end)

        lastRes = res

        if ok and type(res) == "table" and res.Success then

            return true, res

        end

        local message = ok and type(res) == "table" and tostring(res.Message or "") or ""

        -- Retry only when the server told us it was rate limiting (or gave nothing back).
        if res ~= nil and not message:find("Slow down", 1, true) then

            return false, res

        end

        task.wait(0.55)

    end

    return false, lastRes

end

local function EquipBestTeam()

    -- One odd inventory entry used to take the whole pick down with it.
    local function safeBestUnits()

        local ok, ids = pcall(GetBestUnits)

        if ok and type(ids) == "table" then return ids end

        return {}

    end

    local topIds = safeBestUnits()

    -- Inventory can still be replicating right after a join or respawn.
    if #topIds == 0 then

        task.wait(0.4)

        topIds = safeBestUnits()

    end

    if #topIds == 0 then return false end

    return (SetTeam(topIds))

end

    return bestBySlot, allOwnedGears

end



local function EquipBestGear()

    local bestBySlot = GetBestGears()

    if not Remotes then return false, false end

    local equipRemote = Remotes.Get("Items", "Equip")

    if not equipRemote then return false, false end

    

    local anyEquipped = false

    local equippedCount = 0

    for slot, bestGear in pairs(bestBySlot) do

        local cur = LocalPlayer and LocalPlayer:GetAttribute("Gear" .. slot)

        if cur ~= bestGear.Id then

            local ok = pcall(function()

                equipRemote:Invoke({

                    Slot = slot,

                    Id = bestGear.Id

                })

            end)

            if ok then

                anyEquipped = true

                equippedCount = equippedCount + 1

            end

            task.wait(0.1)

        end

    end

    return true, anyEquipped, equippedCount

end

    if IsInDungeon() then

        DungeonEngineStatus = "กำลังต่อสู้ใน Double Dugeons (เปิด Auto โจมตีอยู่)..."

        pcall(function()

                ClickGuiButton(auto)

                EquipBestGear()

            end

        end)

        return

    end



    -- 2. คูลดาวน์ระหว่างรอบ

    if os.clock() - lastDungeonAttempt < 3.0 then

        DungeonEngineStatus = "รอคูลดาวน์ก่อนเริ่มรอบใหม่..."

        return

    end



    -- 3. ตรวจสอบเงื่อนไขตัวละคร (ไม่ติดถือการ์ด)

    if IsHoldingPack and IsHoldingPack() then

        DungeonEngineStatus = "รอส่งการ์ดที่ถือก่อนเข้า Double Dugeons..."

        return

    end



    -- ลงจากลู่วิ่งหากกำลังวิ่งอยู่ (ป้องกัน error CantNow)

    if IsOnTreadmill and IsOnTreadmill() then

        DungeonEngineStatus = "กำลังลงจากลู่วิ่งเพื่อเตรียมเข้า Double Dugeons..."

        ExitTreadmill()

        task.wait(0.5)

        if IsOnTreadmill() then return end

    end



    -- 4. ตรวจสอบ Speed (ต้องการ 50B+)

    local speed = tonumber(PlayerData and PlayerData.Get and PlayerData.Get("Speed")) or 0

    local minSpeed = (TowerConfig and TowerConfig.Portal and TowerConfig.Portal.MinimumSpeed) or 50000000000

    if speed < minSpeed then

        local spdStr = NumberFormatter and NumberFormatter.FormatCompact(speed) or tostring(speed)

        DungeonEngineStatus = string.format("Speed ยังไม่ถึง 50B (มี %s / 50B)", spdStr)

        return

    end



    -- 5. ตรวจสอบสถานะเปิดของประตูพอร์ทัล

    if DungeonUtil and DungeonUtil.GetPortal then

        local isOpen, timeLeft = DungeonUtil.GetPortal(workspace:GetServerTimeNow())

        if not isOpen then

            DungeonEngineStatus = string.format("รอประตู Double Dugeons เปิด (เหลืออีก ~%ds)...", math.max(0, math.floor(timeLeft or 0)))

            return

        end

    end



    -- 6. จัดทีมตัวละครและสวมใส่เกียร์ที่ดีที่สุดก่อนเริ่ม

    if Config.AutoEquipBestTeam then

        EquipBestTeam()

        task.wait(0.2)

    end

    if Config.AutoEquipBestGear then

        EquipBestGear()

        task.wait(0.15)

    end



    -- 7. เปิดปาร์ตี้ Double Dugeons และเริ่มเกม

    DungeonEngineStatus = "กำลังเปิดปาร์ตี้ Double Dugeons..."

    lastDungeonAttempt = os.clock()



    local okOpen, resOpen = pcall(function()

        return Remotes.Get("DungeonParty", "Open"):Invoke("Double Dugeons")

    end)



    if okOpen and type(resOpen) == "table" and resOpen.Success then

        DungeonEngineStatus = "เปิดปาร์ตี้สำเร็จ! กำลังเริ่มดันเจี้ยน..."

        task.wait(0.3)

        local okStart, resStart = pcall(function()

            return Remotes.Get("DungeonParty", "Start"):Invoke()

        end)

        if okStart and type(resStart) == "table" and resStart.Success then

            DungeonEngineStatus = "เข้าสู่ Double Dugeons สำเร็จ! กำลังต่อสู้..."

            task.wait(0.5)

            pcall(function()

                Remotes.Get("Towers", "SetAuto"):Fire(true)

            end)

        else

            local reason = (okStart and type(resStart) == "table" and resStart.Reason) or "Failed"

            if reason == "CantNow" or reason == "Busy" then

                DungeonEngineStatus = "ติดสถานะเซิร์ฟเวอร์ (" .. tostring(reason) .. ") กำลังเคลียร์ปาร์ตี้..."

                ExitCurrentDungeon()

            else

                DungeonEngineStatus = "เริ่มดันเจี้ยนไม่สำเร็จ (" .. tostring(reason) .. ") รอสักครู่..."

            end

        end

    else

        local reason = (okOpen and type(resOpen) == "table" and (resOpen.Message or resOpen.Reason)) or "Failed"

        if reason == "Pick at least one unit!" then

            DungeonEngineStatus = "กำลังจัดทีมตัวละครใหม่..."

            EquipBestTeam()

        elseif reason == "CantNow" or reason == "Busy" then

            DungeonEngineStatus = "ติดสถานะปาร์ตี้เดิม กำลังรีเซ็ต..."

            ExitCurrentDungeon()

        else

            DungeonEngineStatus = "เปิดปาร์ตี้ไม่สำเร็จ (" .. tostring(reason) .. ") รอสักครู่..."

        end

    end

end

local workerThread = nil



local function DepositPack(stageName)

    if not IsHoldingPack() then return end

    Config.Status = "กำลังนำส่งที่จุดเตรียมตัว/Safe Zone (1155, 71, -363)..."



    local hrp, hum = GetCharParts()

    if not hrp or not hum or hum.Health <= 0 then return end



    StopGroundMovement()



    -- Routing directly through staging point (1155, 71, -363)

    local targetPos = STAGING_POSITION



    local initialDist = (Vector3.new(targetPos.X, 0, targetPos.Z) - Vector3.new(hrp.Position.X, 0, hrp.Position.Z)).Magnitude

    local speed = GetSafeMaxSpeed()

    local dynamicTimeout = math.max((initialDist / math.max(speed, 25)) + 6, 25)

    local t0 = os.clock()

    local reached = false



    activeGroundConn = RunService.Heartbeat:Connect(function()

        if not Config.IsFarmRunning or not hrp or not hrp.Parent or hum.Health <= 0 then

            reached = true

            StopGroundMovement()

            return

        end



        if IsInDungeon and IsInDungeon() then

            reached = true

            StopGroundMovement()

            return

        end



        if not IsHoldingPack() then

            reached = true

            StopGroundMovement()

            return

        end



        if LocalPlayer.Character and LocalPlayer.Character:GetAttribute("Ragdolled") == true then

            RecoverFromRagdoll(LocalPlayer.Character)

        end



        local curPos = hrp.Position



        if curPos.Y < MIN_SAFE_Y then

            local safeZ = curPos.Z

            if IsInCorridor(curPos) then

                safeZ = math.clamp(curPos.Z, CORRIDOR_Z_MIN, CORRIDOR_Z_MAX)

            end

            hrp.CFrame = CFrame.new(curPos.X, DEFAULT_GROUND_Y, safeZ)

            hrp.AssemblyLinearVelocity = Vector3.zero

        end



        local delta2D = Vector3.new(targetPos.X - curPos.X, 0, targetPos.Z - curPos.Z)

        local dist2D = delta2D.Magnitude



        -- Safe Zone / Staging reached (1155, 71, -363)

        if dist2D <= 3.0 or curPos.X <= 1155.0 or (os.clock() - t0 > dynamicTimeout) then

            reached = true

            StopGroundMovement()

            return

        end



        -- Same distance-aware braking while hauling a card (full speed until close).
        speed = math.min(GetSafeMaxSpeed(), math.max(dist2D * 10, 8))

        local dir2D = delta2D.Unit



        -- Smooth wall deflection & multi-enemy avoidance

        if IsInCorridor(curPos) then

            local evadeVec = GetMultiEnemyAvoidance(curPos)

            if evadeVec.Magnitude > 0 then

                local testZ = curPos.Z + (dir2D.Z + evadeVec.Z) * 6

                if testZ > (CORRIDOR_Z_MAX - 8) or testZ < (CORRIDOR_Z_MIN + 8) then

                    evadeVec = Vector3.new(0, 0, -evadeVec.Z)

                end

                dir2D = (dir2D + evadeVec).Unit

            end



            if curPos.Z > (CORRIDOR_Z_MAX - 6) and dir2D.Z > 0 then

                dir2D = Vector3.new(dir2D.X, 0, -0.6).Unit

            elseif curPos.Z < (CORRIDOR_Z_MIN + 6) and dir2D.Z < 0 then

                dir2D = Vector3.new(dir2D.X, 0, 0.6).Unit

            end

        end



        hum.WalkSpeed = speed

        hum:Move(dir2D, false)

        -- The engine's PlayerModule rewrites MoveDirection every frame, so
        -- Humanoid:Move alone never sticks. The velocity write is what actually
        -- moves us — kept at the game's own walk speed so nothing corrects it.

        local currentYVel = math.max(hrp.AssemblyLinearVelocity.Y, -8)

        hrp.AssemblyLinearVelocity = Vector3.new(dir2D.X * speed, currentYVel, dir2D.Z * speed)

    end)

    getgenv()._AnimeGroundConn = activeGroundConn



    while not reached and Config.IsFarmRunning and IsHoldingPack() and hrp and hum.Health > 0 and (os.clock() - t0 <= dynamicTimeout) do

        task.wait(0.02)

    end



    StopGroundMovement()



    if IsHoldingPack() then

        local waitStart = os.clock()

        while IsHoldingPack() and Config.IsFarmRunning and (os.clock() - waitStart < 4) do

            task.wait(0.1)

        end

        task.wait(0.2)

        TryAutoSellSelectedCards()

        if Config.AutoUpgradeTreadmill then

            TryUpgradeTreadmill()

        end

        if Config.AutoRebirth then

            TryRebirth()

        end

    end

end



-- Reliable Prompt Steal with Full Hold Completion & Server Replication Grace

local function StealPack(target, stageName)

    local prompt = target.Prompt or (target.Pack and target.Pack.Parent and target.Pack:FindFirstChildWhichIsA("ProximityPrompt", true))

    if not prompt or not prompt.Parent or IsHoldingPack() then return end



    local hrp, hum = GetCharParts()

    if not hrp or not hum or hum.Health <= 0 then return end



    -- Solid halt right at card prompt (prevents sliding out of range)

    StopGroundMovement()

    hum:Move(Vector3.zero, false)

    hrp.AssemblyLinearVelocity = Vector3.zero



    local holdDuration = prompt.HoldDuration or 0.2



    -- 1. Dual prompt fire: UNC native instant fire + Engine UI hold

    pcall(function() fireproximityprompt(prompt, 30, true) end)

    pcall(function() prompt:InputHoldBegin() end)



    -- 2. Wait full hold duration without premature cancellation

    local tStart = os.clock()

    while os.clock() - tStart < (holdDuration + 0.1) and Config.IsFarmRunning and not IsHoldingPack() do

        local curHrp, curHum = GetCharParts()

        if not curHrp or not curHum or curHum.Health <= 0 then

            pcall(function() prompt:InputHoldEnd() end)

            return

        end

        if LocalPlayer.Character and LocalPlayer.Character:GetAttribute("Ragdolled") == true then

            pcall(function() prompt:InputHoldEnd() end)

            RecoverFromRagdoll(LocalPlayer.Character)

            return

        end

        task.wait(0.04)

    end



    -- 3. Trigger proximity prompt cleanly once

    pcall(function() prompt:InputHoldEnd() end)

    pcall(function() fireproximityprompt(prompt) end)



    -- 4. Server replication grace period: wait up to 1.5s for IsHoldingPack() to confirm

    local tWait = os.clock()

    while not IsHoldingPack() and Config.IsFarmRunning and (os.clock() - tWait < 1.5) do

        local curHrp, curHum = GetCharParts()

        if not curHrp or not curHum or curHum.Health <= 0 then break end

        if LocalPlayer.Character and LocalPlayer.Character:GetAttribute("Ragdolled") == true then

            RecoverFromRagdoll(LocalPlayer.Character)

            break

        end

        task.wait(0.05)

    end

end



local function StartMultiStageWorker()

    if workerThread then

        task.cancel(workerThread)

        workerThread = nil

    end

    if getgenv()._AnimePackFarmWorker then

        pcall(task.cancel, getgenv()._AnimePackFarmWorker)

        getgenv()._AnimePackFarmWorker = nil

    end



    Config.IsFarmRunning = true



    workerThread = task.spawn(function()

        UnequipHeldTools()



        -- เดินไปจุดเตรียมตัว (1155, 71, -363) ตอนเริ่มก่อนเสมอ

        if IsOnTreadmill() then

            ExitTreadmill()

        end

        local initialHrp = GetCharParts()

        if initialHrp and not IsInCorridor(initialHrp.Position) then

            Config.Status = "กำลังเดินไปจุดเตรียมตัว (1155, 71, -363)..."

            GoToStagingPosition("AutoFarmToggle")

        end



        while Config.IsFarmRunning do

            local hrp, hum = GetCharParts()

            if not hrp or not hum or hum.Health <= 0 then

                task.wait(0.5)

                continue

            end



            if IsInDungeon and IsInDungeon() then

                Config.Status = "กำลังลงดันเจี้ยน (พักการฟาร์มการ์ดชั่วคราว)..."

                task.wait(1.0)

                continue

            end



            -- 1. นำการ์ดไปส่งที่ Safe Zone หากถืออยู่ (ผ่านจุด Staging 1155, 71, -363)

            if IsHoldingPack() then

                if IsOnTreadmill() then

                    ExitTreadmill()

                end

                DepositPack()

                if Config.AutoPlaceCards then

                    TryAutoPlaceCards()

                end

                continue

            end



            -- 2. ค้นหาการ์ดที่ดีที่สุดตามระดับที่เลือก จากทุกด่านที่เปิดไว้

            local target = FindBestTargetInActiveStages()



            if target then

                UnequipHeldTools()

                if IsOnTreadmill() then

                    Config.Status = string.format("พบการ์ด %s [%s] ใน %s กำลังลงจากลู่วิ่ง...", target.Name, target.Rarity, target.Stage)

                    ExitTreadmill()

                    task.wait(0.15)

                end



                local curHrp = GetCharParts()

                if curHrp and not IsInCorridor(curHrp.Position) then

                    Config.Status = "ไปจุดเตรียมตัว (1155, 71, -363) ก่อนออกไปเก็บ..."

                    GoToStagingPosition("BeforeDeparture")

                end



                local stealAttempts = 0

                local stealStartTime = os.clock()

                local maxAllowedTime = 0



                while Config.IsFarmRunning and not IsHoldingPack() do

                    local loopHrp, loopHum = GetCharParts()

                    if not loopHrp or not loopHum or loopHum.Health <= 0 then

                        task.wait(0.5)

                        continue

                    end



                    if LocalPlayer.Character and LocalPlayer.Character:GetAttribute("Ragdolled") == true then

                        RecoverFromRagdoll(LocalPlayer.Character)

                    end



                    local curPos = loopHrp.Position

                    local packPos = target.Position

                    if target.Prompt and target.Prompt.Parent then

                        if target.Prompt.Parent:IsA("Attachment") then

                            packPos = target.Prompt.Parent.WorldPosition

                        elseif target.Prompt.Parent:IsA("BasePart") then

                            packPos = target.Prompt.Parent.Position

                        elseif target.Prompt.Parent:IsA("Model") then

                            local ok, pPos = pcall(function() return target.Prompt.Parent:GetPivot().Position end)

                            if ok and pPos then packPos = pPos end

                        end

                    elseif target.Pack and target.Pack.Parent then

                        local ok, pPos = pcall(function() return target.Pack:GetPivot().Position end)

                        if ok and pPos then packPos = pPos end

                    end



                    if IsInCorridor(packPos) then

                        local targetZ = math.clamp(packPos.Z, CORRIDOR_Z_MIN + 10, CORRIDOR_Z_MAX - 10)

                        packPos = Vector3.new(packPos.X, packPos.Y, targetZ)

                    end



                    local delta2D = Vector3.new(packPos.X - curPos.X, 0, packPos.Z - curPos.Z)

                    local dist2D = delta2D.Magnitude

                    -- Budget scales with the character's actual travel speed, and only
                    -- ever grows: after arriving, the remaining steal retries still
                    -- need the time banked from the long leg.
                    local travelSpeed = math.max(GetSafeMaxSpeed(), 25)
                    maxAllowedTime = math.max(maxAllowedTime, (dist2D / travelSpeed) * 3 + 10, 30)



                    local targetValid = target and target.Pack and target.Pack.Parent

                    if not targetValid or (os.clock() - stealStartTime > maxAllowedTime) or (stealAttempts >= 4) then

                        if target and target.Pack then

                            TargetBlacklist[target.Pack] = os.clock()

                        end

                        break

                    end



                    -- Refresh prompt & position when getting in streaming range

                    local prompt = target.Pack:FindFirstChild("StealPrompt", true) 

                        or target.Pack:FindFirstChildWhichIsA("ProximityPrompt", true)

                    if prompt then

                        target.Prompt = prompt

                    end



                    -- เดินไปชิดการ์ดในระยะ 2.5 studs อย่างนุ่มนวล (ไม่กระตุก ไม่ snap CFrame)

                    if dist2D > 2.8 then

                        Config.Status = string.format("กำลังเดินไปเอา %s [%s] ใน %s...", target.Name, target.Rarity, target.Stage)

                        GroundGlideTo(packPos, 2.5, target.Stage)

                    end



                    if Config.IsFarmRunning and not IsHoldingPack() then

                        prompt = target.Pack:FindFirstChild("StealPrompt", true) 

                            or target.Pack:FindFirstChildWhichIsA("ProximityPrompt", true)

                        if prompt and prompt.Parent then

                            target.Prompt = prompt

                            Config.Status = string.format("กำลังขโมย %s ใน %s...", target.Name, target.Stage)

                            stealAttempts = stealAttempts + 1

                            StealPack(target, target.Stage)

                        else

                            stealAttempts = stealAttempts + 1

                            task.wait(0.2)

                        end

                    end



                    if IsHoldingPack() then

                        break

                    end



                    task.wait(0.06)

                end



                if IsHoldingPack() then

                        TryAutoPlaceCards()

                    end

                end

            else

                -- ไม่มีการ์ดในด่านที่เลือก / เก็บหมดแล้ว:

                if IsOnTreadmill() then

                    Config.Status = "กำลังฝึกซ้อมบนลู่วิ่ง รอการ์ดเกิดใหม่..."

                    if Config.AutoUpgradeTreadmill then

                        TryUpgradeTreadmill()

                    end

                    task.wait(0.4)

                else

                    local curHrp = GetCharParts()

                    local distToStaging = curHrp and (Vector3.new(STAGING_POSITION.X, 0, STAGING_POSITION.Z) - Vector3.new(curHrp.Position.X, 0, curHrp.Position.Z)).Magnitude or 999



                    -- วิ่งกลับ position ก่อนเริ่ม (1155, 71, -363)

                        GoToStagingPosition("NoCardsReturn")

                    end



                    -- เมื่อกลับถึงจุดเริ่มต้นแล้ว เข้าลู่วิ่ง

                    if Config.AutoTreadmillWhenEmpty then

                        Config.Status = "กลับถึงจุดเริ่มต้นแล้ว กำลังขึ้นลู่วิ่งที่ฐาน..."

                        end

                        EnterTreadmill()

                    else

                        Config.Status = "ไม่มีการ์ด รอการ์ดเกิดใหม่ที่จุดเริ่มต้น..."

                    end

                    task.wait(0.3)

                end

            end



            task.wait(0.1)

        end



        StopGroundMovement()

        if IsOnTreadmill() then

        end

        Config.Status = "Idle"

    end)

local CodeRedeemRunning = false

-- Codes already claimed by this account are stored in PlayerData.RedeemedCodes.
-- It can be a set (code = true) or a plain array, so both shapes are checked.
local function IsCodeRedeemed(code)
    if not PlayerData or not PlayerData.Get then return false end

    local redeemed = PlayerData.Get("RedeemedCodes")
    if type(redeemed) ~= "table" then return false end
    if redeemed[code] == true then return true end

    for _, value in pairs(redeemed) do
        if type(value) == "string" and value == code then return true end
    end

    return false
end

local function RedeemCodeOnce(code)
    if not Remotes then return false, "Remotes ไม่พร้อม" end

    local ok, result = pcall(function()
        return Remotes.Get("Codes", "Redeem"):Invoke(code)
    end)

    if not ok then return false, "เรียกไม่สำเร็จ" end

    if type(result) == "table" then
        local message = result.Message
        if type(message) ~= "string" then
            message = result.Success and "redeemed" or "ไม่สำเร็จ"
        end
        return result.Success == true, message
    end

    return false, "ไม่มีข้อมูลตอบกลับ"
end

local function RedeemAllCodes()
    if CodeRedeemRunning then
        if WindUI and WindUI.Notify then
            WindUI:Notify({ Title = "Redeem All Codes", Content = "กำลังรับโค้ดอยู่ กรุณารอให้จบก่อน", Duration = 3 })
        end
        return
    end

    CodeRedeemRunning = true

    task.spawn(function()
        local successCount, failedCount, skipped = 0, 0, 0
        local failedCodes = {}

        for _, code in ipairs(HARDCODED_CODES) do
            if IsCodeRedeemed(code) then
                skipped = skipped + 1
            else
                local ok, message = RedeemCodeOnce(code)

                if ok then
                    successCount = successCount + 1
                    print(string.format("[Codes] OK   %s -> %s", code, message))
                else
                    failedCount = failedCount + 1
                    table.insert(failedCodes, code .. " (" .. tostring(message) .. ")")
                    print(string.format("[Codes] FAIL %s -> %s", code, message))
                end

                task.wait(1.2) -- server rate limit is 0.5s; 1.2s keeps the queue clean
            end
        end

        CodeRedeemRunning = false

        local summary = string.format(
            "สำเร็จ %d | ไม่ผ่าน %d | ข้ามเพราะเคยรับแล้ว %d (ทั้งหมด %d)",
            successCount, failedCount, skipped, #HARDCODED_CODES
        )

        print("[Codes] " .. summary)
        if #failedCodes > 0 then
            print("[Codes] ไม่ผ่าน: " .. table.concat(failedCodes, ", "))
        end

        if WindUI and WindUI.Notify then
            WindUI:Notify({
                Title = successCount > 0 and "Redeem Codes สำเร็จ" or "Redeem Codes เสร็จสิ้น",
                Content = summary,
                Duration = 5,
            })
        end
    end)
end

local ICOnDefeat = nil

-- Iterative search — HUD trees are deep and GetDescendants allocates too much.
local function ICDeepFind(root, name, className)
    if not root then return nil end
    local stack = { root }
    local visited = 0
    while #stack > 0 and visited < 800 do
        local node = table.remove(stack)
        visited = visited + 1
        if node.Name == name and (not className or node:IsA(className)) then
            return node
        end
        for _, child in ipairs(node:GetChildren()) do
            stack[#stack + 1] = child
        end
    end
    return nil
end

local ICHud = {
    hud = nil,
    floorLabel = nil,
    waveLabel = nil,
    wavesLabel = nil,
    timerLabel = nil,
    startsLabel = nil,
    defeatFrame = nil,
    replayButton = nil,
    defeatConn = nil,
}

local function ICRefreshHud()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    local hud = pg and pg:FindFirstChild("DungeonHud")
    if hud == ICHud.hud then return hud end

    if ICHud.defeatConn then
        pcall(function() ICHud.defeatConn:Disconnect() end)
        ICHud.defeatConn = nil
    end
    if ICHud.victoryConn then
        pcall(function() ICHud.victoryConn:Disconnect() end)
        ICHud.victoryConn = nil
    end

    ICHud.hud = hud
    ICHud.floorLabel = nil
    ICHud.waveLabel = nil
    ICHud.wavesLabel = nil
    ICHud.timerLabel = nil
    ICHud.startsLabel = nil
    ICHud.defeatFrame = nil
    ICHud.replayButton = nil

    if not hud then return nil end

    local top = hud:FindFirstChild("TopBar")
    local banner = top and top:FindFirstChild("Banner")
    ICHud.floorLabel = banner and banner:FindFirstChild("Floor")
    ICHud.waveLabel = banner and banner:FindFirstChild("wAVE")

    for _, node in ipairs(hud:GetDescendants()) do
        if node:IsA("TextLabel") then
            local text = tostring(node.Text)
            if not ICHud.wavesLabel and string.find(text, "Waves Completed") then
                ICHud.wavesLabel = node
            elseif not ICHud.timerLabel and string.find(text, "Total Time") then
                ICHud.timerLabel = node
            elseif not ICHud.startsLabel and string.find(string.upper(text), "STARTS IN") then
                ICHud.startsLabel = node
            end
        end
    end

    ICHud.defeatFrame = ICDeepFind(hud, "Defeat", "GuiObject")
    if ICHud.defeatFrame then
        ICHud.replayButton = ICDeepFind(ICHud.defeatFrame, "Replay", "GuiButton")
        ICHud.defeatConn = ICHud.defeatFrame:GetPropertyChangedSignal("Visible"):Connect(function()
            if ICHud.defeatFrame and ICHud.defeatFrame.Visible and ICOnDefeat then
                ICOnDefeat()
            end
        end)
    end

    -- Victory / clear modal uses the same Replay flow (player keeps farming floors).
    local victoryFrame = ICDeepFind(hud, "Victory", "GuiObject") or ICDeepFind(hud, "Win", "GuiObject")
    if victoryFrame then
        if not ICHud.replayButton then
            ICHud.replayButton = ICDeepFind(victoryFrame, "Replay", "GuiButton")
        end
        ICHud.victoryConn = victoryFrame:GetPropertyChangedSignal("Visible"):Connect(function()
            if victoryFrame.Visible and ICOnDefeat then
                ICOnDefeat()
            end
        end)
    end

    return hud
end

local ICStartScanAt = 0

-- The countdown label is created per run, so it cannot be cached at startup.
local function ICScanStartsLabel()
    if ICHud.startsLabel and ICHud.startsLabel.Parent then return ICHud.startsLabel end
    ICHud.startsLabel = nil
    if os.clock() - ICStartScanAt < 2 then return nil end
    ICStartScanAt = os.clock()
    local hud = ICHud.hud
    if not hud then return nil end
    for _, node in ipairs(hud:GetDescendants()) do
        if node:IsA("TextLabel") and string.find(string.upper(tostring(node.Text)), "STARTS IN") then
            ICHud.startsLabel = node
            return node
        end
    end
    return nil
end

local function ICIsInCastle()
    if IC_PLACE_IDS[game.PlaceId] then return true end
    if workspace:FindFirstChild("InfinityCastle") then return true end
    local label = ICHud.floorLabel
    if label and string.find(string.upper(tostring(label.Text)), "INFINITY") then return true end
    return false
end

-- Section 11.6 reads this global so both engines never run at the same time.
getgenv().IsInInfinityCastle = ICIsInCastle

local function ICReadProgress()
    ICRefreshHud()
    IC.InCastle = ICIsInCastle()

    local waveLabel = ICHud.waveLabel
    local waveText = waveLabel and tostring(waveLabel.Text) or ""
    local wave, left = string.match(waveText, "[Ww][Aa][Vv][Ee]%s*(%d+)%s*[^%d]*(%d+)%s*[Ll][Ee][Ff][Tt]")
    if not wave then wave = string.match(waveText, "[Ww][Aa][Vv][Ee]%s*(%d+)") end
    if wave then
        IC.Wave = tonumber(wave) or IC.Wave
        IC.LastWaveAt = os.clock()
        if IC.Wave > IC.BestWave then IC.BestWave = IC.Wave end
    end
    if left then IC.EnemiesLeft = tonumber(left) or IC.EnemiesLeft end

    if ICHud.wavesLabel then
        local completed = tonumber(string.match(tostring(ICHud.wavesLabel.Text), "(%d+)"))
        if completed then IC.WavesCompleted = completed end
    end

    if ICHud.timerLabel then IC.TimerText = tostring(ICHud.timerLabel.Text) end

    local startsLabel = ICScanStartsLabel()
    if startsLabel then
        IC.StartsIn = tonumber(string.match(tostring(startsLabel.Text), "(%d+)"))
    end
end

local function ICFire(namespace, name, ...)
    if not Remotes then return false end
    local args = { ... }
    local ok = pcall(function()
        local remote = Remotes.Get(namespace, name)
        if not remote then return end
        remote:Fire(table.unpack(args))
    end)
    return ok
end

local function ICInvoke(namespace, name, ...)
    if not Remotes then return false, nil end
    local args = { ... }
    local ok, result = pcall(function()
        local remote = Remotes.Get(namespace, name)
        if not remote then return nil end
        return remote:Invoke(table.unpack(args))
    end)
    if not ok then return false, nil end
    return true, result
end

    return {}
end

local function ICStartRun()
    if ICIsInCastle() then return false, "อยู่ในปราสาทอยู่แล้ว" end
    if IsInDungeon and IsInDungeon() and not ICIsInCastle() then return false, "อยู่ในดันเจี้ยนอื่นอยู่" end

    -- Fixed cooldown between entry attempts (server rate limit is 0.5s; 5s is safe).
    local IC_REQUEUE_SECONDS = 5
    if os.clock() - IC.LastAttempt < IC_REQUEUE_SECONDS then
        local waitLeft = math.max(1, IC_REQUEUE_SECONDS - math.floor(os.clock() - IC.LastAttempt))
        return false, string.format("รอคูลดาวน์อีก %ds", waitLeft)
    end
    IC.LastAttempt = os.clock()

    if IsOnTreadmill and IsOnTreadmill() then
        pcall(ExitTreadmill)
        task.wait(0.4)
    end

    if Config.AutoEquipBestTeam and EquipBestTeam then
        pcall(EquipBestTeam)
        task.wait(0.2)
    end
    if Config.AutoEquipBestGear and EquipBestGear then
        pcall(EquipBestGear)
        task.wait(0.15)
    end

    local ids = ICGetTeamIds()
    if #ids == 0 then return false, "ไม่มีตัวละครในทีม (Pick at least one unit)" end

    -- Path 1 — party flow. Order matters: SetTeam -> Open -> Start (verified live).
    local okTeam, teamResult = SetTeam(ids)
    local teamOk = okTeam and not (type(teamResult) == "table" and teamResult.Success == false)
    if teamOk then
        task.wait(0.6)

        for _, towerName in ipairs(IC_TOWER_NAMES) do
            local openResult = nil
            -- The server rate-limits at 0.5s; one retry covers a "Slow down!" kick.
            for attempt = 1, 2 do
                local okOpen, result = ICInvoke("DungeonParty", "Open", towerName)
                if okOpen and type(result) == "table" then
                    openResult = result
                    if result.Success or result.Message ~= "Slow down!" then break end
                elseif okOpen then
                    openResult = nil
                    break
                end
                task.wait(0.7)
            end

            if type(openResult) == "table" and openResult.Success then
                task.wait(0.6)
                local okStart, startResult = ICInvoke("DungeonParty", "Start")
                if okStart and type(startResult) == "table" and startResult.Success then
                    IC.Runs = IC.Runs + 1
                    IC.KnownReady = false
                    return true, "เข้าปราสาทสำเร็จ (" .. towerName .. ")"
                end
                local startMessage = (okStart and type(startResult) == "table" and startResult.Message) or "Start ไม่ตอบสนอง"
                return false, "เปิดปาร์ตี้แล้วแต่เริ่มไม่สำเร็จ: " .. tostring(startMessage)
            end

            if type(openResult) == "table" and openResult.Message then
                local message = tostring(openResult.Message)
                if message ~= "Pick at least one unit!" and message ~= "Slow down!" then
                    return false, "เปิดปาร์ตี้ไม่สำเร็จ: " .. message
                end
            end

            task.wait(0.5)
        end
    end

    -- Path 2 — direct tower payload (only answers inside the dungeon place).
    for _, towerName in ipairs(IC_TOWER_NAMES) do
        local ok, result = ICInvoke("Towers", "Start", { Tower = towerName, Ids = ids, Auto = true })
        if ok and type(result) == "table" and result.Success then
            IC.Runs = IC.Runs + 1
            IC.KnownReady = false
            return true, "เข้าปราสาทสำเร็จ (" .. towerName .. ")"
        end
        task.wait(0.3)
    end

    -- Path 3 — UI click fallback.
    local okClick, clicked = pcall(ICClickTowerUi)
    if okClick and clicked then
        IC.Runs = IC.Runs + 1
        IC.KnownReady = false
        return true, "กดปุ่มเริ่มจาก UI Towers สำเร็จ"
    end

    return false, "เริ่มไม่สำเร็จ (จะลองใหม่รอบถัดไป)"
end

local function ICReplayRun()
    if os.clock() - IC.LastReplay < 3 then return false end
    IC.LastReplay = os.clock()

    local ok = ICFire("Towers", "Replay")
    if not ok then ok = ICFire("Towers", "Restart") end
    if not ok and ICHud.replayButton then
        ClickGuiButton(ICHud.replayButton)
        ok = true
    end

    if ok then
        IC.Runs = IC.Runs + 1
        IC.KnownReady = false
    end
    return ok
end

-- Drops spawn as loose parts in Workspace.Drops; touching collects them.
local function ICCollectDrops()
    if os.clock() - IC.LastCollect < 0.4 then return end
    IC.LastCollect = os.clock()

    local drops = workspace:FindFirstChild("Drops")
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    if not drops or not hrp then return end

    local collected = 0
    local visited = 0
    for _, part in ipairs(drops:GetChildren()) do
        visited = visited + 1
        if visited > 40 then break end
        if part:IsA("BasePart") then
            hrp.CFrame = CFrame.new(part.Position + Vector3.new(0, 1.5, 0))
            collected = collected + 1
            task.wait(0.06)
        end
    end

    if collected > 0 then
        IC.DropsCollected = IC.DropsCollected + collected
    end
end

local function ICInventoryPacks()
    local packs = {}
    if not PlayerData or not PlayerData.Get then return packs end
    local inventory = PlayerData.Get("Inventory")
    if type(inventory) ~= "table" or type(inventory.Packs) ~= "table" then return packs end

    for _, bucket in ipairs({ inventory.Packs.Placed, inventory.Packs.Backpack }) do
        if type(bucket) == "table" then
            for _, pack in pairs(bucket) do
                if type(pack) == "table" and pack.Id then
                    packs[pack.Id] = pack
                end
            end
        end
    end
    return packs
end

-- Inventory diff -> rare-drop counter (Infinity Castle only).
local function ICScanLoot()
    if os.clock() - IC.LastLootScan < 5 then return end
    IC.LastLootScan = os.clock()

    local packs = ICInventoryPacks()
    if not IC.KnownReady then
        IC.KnownPacks = packs
        IC.KnownReady = true
        return
    end

    local rank = PacksConfig and PacksConfig.RarityRank
    local minRank = (rank and rank[Config.WebhookMinRarity]) or 6

    for id, pack in pairs(packs) do
        if not IC.KnownPacks[id] then
            local rarity = pack.Rarity or "Common"
            local packRank = (rank and rank[rarity]) or 1
            if packRank >= minRank then
                IC.RareDrops = IC.RareDrops + 1
            end
        end
    end

    IC.KnownPacks = packs
end

ICOnDefeat = function()
    if not Config.AutoInfinityCastle and not Config.ICAutoReplay then return end
    IC.Status = "แพ้แล้ว — กำลังเริ่มรอบใหม่..."
    if Config.ICAutoReplay then
        task.delay(1.5, function()
            if Config.AutoInfinityCastle or Config.ICAutoReplay then
                pcall(ICReplayRun)
            end
        end)
    end
end

local function ICStep()
    ICReadProgress()

    -- Exit gates are evaluated before anything else so they win over replay.
    if IC.InCastle then
        local waveIsLive = (os.clock() - (IC.LastWaveAt or 0)) < 5
        local stopWave = tonumber(Config.ICStopWaveNumber) or 0
        if Config.ICStopAtWave and stopWave > 0 and waveIsLive and IC.Wave >= stopWave then
            IC.Status = string.format("ถึง Wave %d ตามที่ตั้งไว้ — ออกจากปราสาท", IC.Wave)
            ICExitCastle()
            return
        end
    end

    if not Config.AutoInfinityCastle then
        IC.Status = "ปิดใช้งาน (Idle)"
        return
    end

    if IC.InCastle then
        if Config.ICAutoSkipIntro then ICSkipIntro() end
        ICPushAutoAttack()
        if Config.ICAutoCollectDrops then ICCollectDrops() end
        IC.Status = string.format("กำลังต่อสู้ • Wave %d • เหลือ %d ตัว", IC.Wave, IC.EnemiesLeft)
        return
    end

    -- Not inside the castle: only queue when allowed.
    if not Config.ICAutoEnter then
        IC.Status = "อยู่นอกปราสาท (ปิด Auto Enter)"
        return
    end

    if IC.StartsIn and IC.StartsIn > 0 then
        IC.Status = string.format("รอเริ่มรอบถัดไป (%ds)", IC.StartsIn)
        return
    end

    local ok, message = ICStartRun()
    IC.Status = message
end
